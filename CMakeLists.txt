cmake_minimum_required(VERSION 3.20)
project(diophantasmagoria VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(POLICY CMP0167)
  cmake_policy(SET CMP0167 OLD)
endif()
find_package(Boost 1.74 REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_SOURCE_DIR} OUTPUT_VARIABLE REV OUTPUT_STRIP_TRAILING_WHITESPACE)
file(GLOB SOURCE_FILES CONFIGURE_DEPENDS "src/*.cpp" "src/*.hpp")
set(SOURCE_DIGESTS "")
foreach(SOURCE_FILE IN LISTS SOURCE_FILES)
  file(SHA256 "${SOURCE_FILE}" SOURCE_DIGEST)
  string(APPEND SOURCE_DIGESTS "${SOURCE_DIGEST}")
endforeach()
string(SHA256 SOURCE_ID "${SOURCE_DIGESTS}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${SOURCE_FILES})
option(SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(core src/math.cpp src/io.cpp src/search.cpp src/planning.cpp)
target_include_directories(core PUBLIC src ${Boost_INCLUDE_DIRS})
target_link_libraries(core PUBLIC OpenSSL::Crypto Threads::Threads)
target_compile_definitions(core PUBLIC BUILD_REV="${REV}" BUILD_COMPILER="${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}" BUILD_SOURCE="${SOURCE_ID}" BUILD_TYPE="${CMAKE_BUILD_TYPE}" BOOST_BIND_GLOBAL_PLACEHOLDERS)
target_compile_options(core PRIVATE -Wall -Wextra -Wconversion -Wshadow)
if(SANITIZE)
  target_compile_options(core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(diophantasmagoria src/main.cpp)
target_link_libraries(diophantasmagoria PRIVATE core)
add_executable(unit_tests tests/unit.cpp)
target_link_libraries(unit_tests PRIVATE core)
enable_testing()
add_test(NAME unit COMMAND unit_tests)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:diophantasmagoria>)

add_executable(fault_search tests/fault_search.cpp)
target_link_libraries(fault_search PRIVATE core)
add_test(NAME epochs COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/epochs.py $<TARGET_FILE:diophantasmagoria> $<TARGET_FILE:fault_search>)

add_executable(symmetric_unit_tests tests/symmetric_unit.cpp)
target_link_libraries(symmetric_unit_tests PRIVATE core)
add_test(NAME symmetric_math COMMAND symmetric_unit_tests)
add_test(NAME symmetric_backend COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/symmetric_backend.py $<TARGET_FILE:diophantasmagoria> $<TARGET_FILE:fault_search>)
